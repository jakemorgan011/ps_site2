const SHAPE_PADDING = 48;
const MIN_SLOT_WIDTH = 60;
const MAX_CONTENT_WIDTH = 680; // max text column width, feels like a printed page

const MAX_SPANS = 800;

function getLayout() {
  const contentWidth = Math.min(window.innerWidth - MIN_H_PAD * 2, MAX_CONTENT_WIDTH);
  const contentLeft  = Math.round((window.innerWidth - contentWidth) / 2);
  return { contentWidth, contentLeft };
}

function makeEqCell(tex, caption) {
  const cell = document.createElement('div');
  // flex:1 1 0 + min-width:0 lets the cell shrink below KaTeX's intrinsic width
  // min-width:180px triggers wrapping before content clips
  cell.style.cssText = 'flex:1 1 180px;min-width:0;text-align:center;overflow:hidden;box-sizing:border-box;padding:0 8px;';
  const mathWrap = document.createElement('div');
  katex.render(tex, mathWrap, { throwOnError: false, displayMode: true });
  cell.appendChild(mathWrap);
  if (caption) {
    const cap = document.createElement('p');
    cap.textContent = caption;
    cap.style.cssText = 'margin-top:8px;font:13px Georgia,serif;color:#888;text-align:center;';
    cell.appendChild(cap);
  }
  return cell;
}

function initEquations() {
  for (const block of blocks) {
    if (block.type === 'eq') {
      const div = document.createElement('div');
      div.style.cssText = 'position:absolute;overflow:hidden;text-align:center;';
      div.appendChild(makeEqCell(block.tex, block.caption));
      textLayer.appendChild(div);
      block.el = div;
    } else if (block.type === 'eqrow') {
      const div = document.createElement('div');
      div.style.cssText = 'position:absolute;overflow:hidden;display:flex;flex-wrap:wrap;align-items:flex-start;gap:16px 0;';
      for (const eq of block.eqs) div.appendChild(makeEqCell(eq.tex, eq.caption));
      textLayer.appendChild(div);
      block.el = div;
    } else if (block.type === 'callout') {
      const div = document.createElement('div');
      div.style.cssText = 'position:absolute;text-align:center;';
      const inner = document.createElement('strong');
      inner.textContent = block.text;
      inner.style.cssText = 'font:bold 26px Georgia,serif;letter-spacing:0.01em;';
      div.appendChild(inner);
      textLayer.appendChild(div);
      block.el = div;
    } else if (block.type === 'image') {
      const div = document.createElement('div');
      const [rw, rh] = block.ratio || [10, 6];
      div.style.cssText = `position:absolute;overflow:hidden;aspect-ratio:${rw}/${rh};`;
      const img = document.createElement('img');
      img.src = block.src;
      img.alt = block.alt || '';
      img.style.cssText = `width:100%;height:100%;object-fit:${block.contain ? 'contain' : 'cover'};display:block;`;
      div.appendChild(img);
      textLayer.appendChild(div);
      block.el = div;
    }
  }
}

let scanlineCache = null; // { rows: Array<[x0,x1][]>, w, h }

async function buildScanlines() {
  const svgEl = document.getElementById('artwork');
  const rect  = svgEl.getBoundingClientRect();
  const w = Math.ceil(rect.width);
  const h = Math.ceil(rect.height);
  if (w < 2 || h < 2) return;

  const svgString = new XMLSerializer().serializeToString(svgEl);
  const blob = new Blob([svgString], { type: 'image/svg+xml' });
  const url  = URL.createObjectURL(blob);

  // takes all that svg shiiii and creates the offset for thetext.
  await new Promise((resolve) => {
    const img = new Image();
    img.onload = () => {
      URL.revokeObjectURL(url);
      const canvas = new OffscreenCanvas(w, h);
      const ctx    = canvas.getContext('2d');
      ctx.drawImage(img, 0, 0, w, h);
      const { data } = ctx.getImageData(0, 0, w, h);

      const rows = [];
      for (let row = 0; row < h; row++) {
        const rowSpans = [];
        let inSpan = false, x0 = 0;
        for (let col = 0; col < w; col++) {
          const alpha = data[(row * w + col) * 4 + 3];
          if (alpha > 8  && !inSpan) { inSpan = true;  x0 = col; }
          if (alpha <= 8 &&  inSpan) { inSpan = false; rowSpans.push([x0, col]); }
        }
        if (inSpan) rowSpans.push([x0, w]);
        rows.push(rowSpans);
      }
      scanlineCache = { rows, w, h };
      resolve();
    };
    img.onerror = () => { URL.revokeObjectURL(url); resolve(); };
    img.src = url;
  });
}

function carveSlots(totalWidth, blocked) {
  let slots = [{ left: 0, right: totalWidth }];
  for (const b of blocked) {
    const next = [];
    for (const slot of slots) {
      if (b.right <= slot.left || b.left >= slot.right) {
        next.push(slot);
      } else {
        if (b.left  > slot.left)  next.push({ left: slot.left, right: b.left });
        if (b.right < slot.right) next.push({ left: b.right,   right: slot.right });
      }
    }
    slots = next;
  }
  return slots.filter(s => s.right - s.left >= MIN_SLOT_WIDTH);
}

function getBlockedForDocY(docY, contentLeft) {
  if (!scanlineCache) return [];
  const rect = document.getElementById('artwork').getBoundingClientRect();
  const t    = (docY - window.scrollY - rect.top) / rect.height;
  const row  = Math.round(t * scanlineCache.h);
  if (row < 0 || row >= scanlineCache.h) return [];
  const scaleX = rect.width / scanlineCache.w;
  return scanlineCache.rows[row].map(([c0, c1]) => ({
    left:  rect.left - contentLeft + c0 * scaleX - SHAPE_PADDING,
    right: rect.left - contentLeft + c1 * scaleX + SHAPE_PADDING,
  }));
}

function render() {
  const { contentWidth, contentLeft } = getLayout();
  let spanIdx = 0;
  let y = 0;

  for (const block of blocks) {
    const { style } = block;
    y += style.marginTop;

    if (block.type === 'eq' || block.type === 'eqrow' || block.type === 'callout' || block.type === 'image') {
      block.el.style.left  = `${contentLeft}px`;
      block.el.style.width = `${contentWidth}px`;
      block.el.style.top   = `${y}px`;
      if (block.type === 'image') {
        const [rw, rh] = block.ratio || [10, 6];
        const h = Math.round(contentWidth * rh / rw);
        block.el.style.height = `${h}px`;
        y += h;
      } else {
        y += block.el.offsetHeight;
      }
    } else {
      const { prepared } = block;
      let cursor = { segmentIndex: 0, graphemeIndex: 0 };

      while (spanIdx < MAX_SPANS - 4) {
        const blocked = getBlockedForDocY(y + style.lineHeight / 2, contentLeft);
        const slots   = carveSlots(contentWidth, blocked);

        if (slots.length === 0) { y += style.lineHeight; continue; }

        let anyLine = false;
        for (const slot of slots) {
          const line = layoutNextLine(prepared, cursor, slot.right - slot.left);
          if (!line) break;

          // yea like what
          const s = spans[spanIdx++];
          s.style.display    = '';
          s.style.font       = style.font;
          s.style.lineHeight = `${style.lineHeight}px`;
          s.style.color      = style.color;
          s.style.left       = `${contentLeft + slot.left}px`;
          s.style.top        = `${y}px`;
          s.textContent      = line.text;
          cursor = line.end;
          anyLine = true;
        }

        if (!anyLine) break;
        y += style.lineHeight;
      }
    }

    y += style.marginBottom;
  }

  for (let i = spanIdx; i < MAX_SPANS; i++) spans[i].style.display = 'none';
  textLayer.style.height = `${y + 80}px`;
}

let rafId = null;
function schedule() {
  if (rafId) return;
  rafId = requestAnimationFrame(() => { rafId = null; render(); });
}

window.addEventListener('scroll', schedule, { passive: true });
window.addEventListener('resize', () => { scanlineCache = null; buildScanlines().then(schedule); });

// KaTeX loads via defer, so wait for DOMContentLoaded before init
window.addEventListener('DOMContentLoaded', () => {
  initEquations();
  buildScanlines().then(() => render());
});
